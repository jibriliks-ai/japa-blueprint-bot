import os, requests
from flask import Flask, request

app = Flask(__name__)

TELEGRAM_TOKEN = os.environ.get("TELEGRAM_BOT_TOKEN")
GEMINI_KEY = os.environ.get("GEMINI_API_KEY")
GEMINI_MODEL = os.environ.get("GEMINI_MODEL", "gemini-3.6-flash")

SYSTEM_PROMPT = "You are Japa Blueprint AI for www.japablueprint.com.ng. Help Nigerians japa for Sweden, Canada, UK, POF, blocked account. Short steps. End with: Full blueprint at www.japablueprint.com.ng"

def ask_gemini(q):
    url = f"https://generativelanguage.googleapis.com/v1beta/models/{GEMINI_MODEL}:generateContent?key={GEMINI_KEY}"
    payload = {"contents":[{"parts":[{"text": SYSTEM_PROMPT + "\n\nUser: " + q}]}]}
    try:
        r = requests.post(url, json=payload, timeout=20)
        data = r.json()
        return data["candidates"][0]["content"]["parts"][0]["text"]
    except Exception as e:
        print(e, data if 'data' in locals() else "")
        return "Try again: Example - Sweden POF?"

@app.route('/webhook', methods=['POST'])
def webhook():
    update = request.get_json()
    if "message" in update and "text" in update["message"]:
        chat_id = update["message"]["chat"]["id"]
        text = update["message"]["text"]
        if text == "/start":
            reply = "Hi! I'm Japa Blueprint AI 🇸🇪🇨🇦🇬🇧 Ask me about Sweden, Canada, UK, POF."
        else:
            reply = ask_gemini(text) + "\n\n📘 www.japablueprint.com.ng"
        requests.post(f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/sendMessage", json={"chat_id": chat_id, "text": reply})
    return "ok"

@app.route('/')
def home():
    return "Japa Bot is live"
