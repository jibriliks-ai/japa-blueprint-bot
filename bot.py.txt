import requests
from telegram import Update
from telegram.ext import ApplicationBuilder, CommandHandler, MessageHandler, filters, ContextTypes

TELEGRAM_TOKEN = "PASTE_YOUR_TELEGRAM_TOKEN_HERE"
GEMINI_KEY = "PASTE_YOUR_GEMINI_KEY_HERE"
GEMINI_MODEL = "gemini-3.6-flash"

SYSTEM_PROMPT = """You are Japa Blueprint AI for www.japablueprint.com.ng. Help Nigerians japa for Sweden, Canada, UK, proof of funds, blocked account, admission, visa. Short practical steps. End with: For full templates visit www.japablueprint.com.ng"""

def ask_gemini(question):
    url = f"https://generativelanguage.googleapis.com/v1beta/models/{GEMINI_MODEL}:generateContent?key={GEMINI_KEY}"
    payload = {"contents": [{"parts": [{"text": SYSTEM_PROMPT + "\n\nUser: " + question}]}]}
    r = requests.post(url, json=payload, timeout=30)
    data = r.json()
    if "error" in data:
        return f"API Error: {data['error']['message']}"
    try:
        return data["candidates"][0]["content"]["parts"][0]["text"]
    except:
        return "Try rephrasing your question. Example: How much is Sweden POF?"

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text("Hi! I'm Japa Blueprint AI 🇸🇪🇨🇦🇬🇧\nAsk me about Sweden, Canada, UK, POF, blocked account, admission, visa.")

async def handle(update: Update, context: ContextTypes.DEFAULT_TYPE):
    q = update.message.text
    await context.bot.send_chat_action(chat_id=update.effective_chat.id, action="typing")
    answer = ask_gemini(q)
    await update.message.reply_text(answer + "\n\n📘 Get Blueprint: www.japablueprint.com.ng")

app = ApplicationBuilder().token(TELEGRAM_TOKEN).build()
app.add_handler(CommandHandler("start", start))
app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle))
print("Bot running...")
app.run_polling()
